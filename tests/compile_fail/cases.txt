### mixed_brands
// Handles from two different pools must not be interchangeable.
const zv = @import("zv");
const Users = zv.Pool(u32, 8, struct {});
const Orders = zv.Pool(u32, 8, struct {});
pub fn main() !void {
    var users: Users = .{};
    var orders: Orders = .{};
    const u = try users.alloc(1);
    _ = orders.get(u);
}

### ranged_literal
// Out-of-range literal is a compile error, not a runtime one.
const zv = @import("zv");
pub fn main() void {
    const pct = zv.Ranged(0, 100).lit(101);
    _ = pct;
}
### expect: literal 101 outside [0, 100]
### index_range
// An index type for 8 elements cannot index a 4-element array, even when the
// index value is only known at runtime. (Plain `arr[i]` compiles: see
// tests/baseline/index_range.zig.)
const zv = @import("zv");
pub fn main() !void {
    const arr = [_]u8{ 1, 2, 3, 4 };
    var raw: u8 = 5;
    _ = &raw;
    const i = try zv.Index(8).from(raw);
    _ = zv.at(arr, i);
}

### ranged_as_too_small
// Converting a wider range into a narrower int is rejected statically, even
// for a runtime value. (Plain `@intCast` compiles: see tests/baseline/narrowing.zig.)
const zv = @import("zv");
pub fn main() !void {
    var raw: u16 = 7;
    _ = &raw;
    const x = try zv.Ranged(0, 1000).from(raw);
    const y: u8 = x.as(u8);
    _ = y;
}
### expect: does not fit u8
